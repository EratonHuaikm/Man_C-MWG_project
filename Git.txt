class DigitalTwin_SELF:
    def __init__(self):
        # Stato fisico interno (Motore Freddo)
        self.rpm_fisico = 0.0
        self.psi_fisico = 300.0       
        self.t_heater_fisico = 75.0   
        self.t_cooler_fisico = 75.0

    def elabora_ciclo_fisico(self, fluido, target_rpm, target_th, target_tc, target_psi, is_online):
        """Simula la cinematica Alfa e la combustione del Metano senza forzature software."""
        
        # 1. LOGICA DI ALIMENTAZIONE (METANO CH4)
        if not is_online:
            # Taglio del Gas: Il bruciatore si spegne, l'energia in ingresso si azzera.
            target_rpm = 0.0
            target_th = 75.0     
            target_psi = 300.0   

        # 2. INERZIA CINEMATICA (ALFA CONFIGURATION)
        if self.rpm_fisico < target_rpm:
            # Accelerazione dipendente dal delta termico (Spool Up)
            coppia_accelerante = max(0.1, (self.t_heater_fisico - self.t_cooler_fisico) / 1000.0)
            self.rpm_fisico += ((target_rpm - self.rpm_fisico) * 0.15 * coppia_accelerante)
        elif self.rpm_fisico > target_rpm:
            # Coasting Down: Decelerazione quadratica reale (Attrito + Pompaggio)
            attrito_cuscinetti = 5.0
            drag_pneumatico = 1.2 * ((self.rpm_fisico / 1000.0) ** 2)
            decelerazione = attrito_cuscinetti + drag_pneumatico
            self.rpm_fisico -= decelerazione
            
            if self.rpm_fisico < target_rpm:
                self.rpm_fisico = max(0.0, target_rpm)
        
        if self.rpm_fisico <= 0.5:
            self.rpm_fisico = 0.0

        # 3. RAFFREDDAMENTO DI NEWTON E COMPRESSIONE (Massa metallo e Gas)
        coeff_termico = 0.08 if target_th > self.t_heater_fisico else 0.015
        self.t_heater_fisico += (target_th - self.t_heater_fisico) * coeff_termico
        self.psi_fisico += (target_psi - self.psi_fisico) * 0.05
        self.t_cooler_fisico += (target_tc - self.t_cooler_fisico) * 0.20

        # Morte Termodinamica
        if self.rpm_fisico == 0.0:
            return 0.0, 0.0, 0.0, round(self.psi_fisico, 1)

        # 4. TERMODINAMICA PURA (NESSUN LIMITE ARTIFICIALE MIN/MAX)
        T_h = (self.t_heater_fisico - 32) * 5/9 + 273.15      
        T_c = (self.t_cooler_fisico - 32) * 5/9 + 273.15      
        P_mean = self.psi_fisico * 6894.76             
        omega = (self.rpm_fisico * 2 * np.pi) / 60              
        
        eta_carnot = 1 - (T_c / max(1.0, T_h))
        R_specific = 4124.0 if fluido == 'Hydrogen' else 2077.0     
        viscosita = 1.05e-5      
        
        rho = P_mean / (R_specific * T_c)
        reynolds_factor = (rho * omega / viscosita) ** (-0.2)
        fattore_attrito = 0.046 * reynolds_factor
        work_pompaggio_perdita = fattore_attrito * rho * (omega**2) * 1.45e-4
        
        ntu_ref = 22.5
        ntu = ntu_ref / (max(0.1, (self.rpm_fisico / 1000.0)) ** 0.45)
        efficacia_rigeneratore = ntu / (ntu + 1.0)
        
        eta_termica = (eta_carnot * efficacia_rigeneratore) - (work_pompaggio_perdita / (P_mean * 2.3e-4 * omega + 1))
        
        # Freno meccanico base
        eta_meccanica_pms = 1.0 - 0.045 
        eta_combinata_base = eta_termica * eta_meccanica_pms * 100.0
        
        # 5. RISONANZA MAGNETICA DI GAUSS (IL PARADOSSO)
        # La risonanza si aggancia naturalmente SOLO a 2000 RPM.
        scostamento_rpm = abs(self.rpm_fisico - 2000.0)
        fattore_risonanza = np.exp(-0.5 * (scostamento_rpm / 15.0)**2)
        
        recupero_induttivo = 8.5 * fattore_risonanza # Dona fino all'8.5% in più se a 2000 RPM precisi
        
        potenza_watt = 0.00385 * P_mean * (T_h - T_c) * (self.rpm_fisico / 60)
        potenza_hp = max(0.0, (potenza_watt - work_pompaggio_perdita) / 745.7)
        
        # Calcolo finale Pura Fisica: Base + Fumi (5%) + Risonanza
        eta_totale_simbiosi = (eta_combinata_base * 1.05) + recupero_induttivo
            
        return round(potenza_hp * 1.01 * 0.0000045, 4), round(max(0.0, eta_totale_simbiosi), 1), round(self.rpm_fisico, 1), round(self.psi_fisico, 1)

class SkenderbegQuantumEdgeCore:
    def __init__(self):
        self.resonance_hz = 433.62
        self.battery_kwh = 0.0
        self.motore_fisico = DigitalTwin_SELF()
        self._carica_memoria_genetica()

    def _carica_memoria_genetica(self):
        if os.path.exists(MEMORIA_FILE):
            try:
                with open(MEMORIA_FILE, "r") as f: self.battery_kwh = json.load(f).get("battery_kwh", 0.0)
            except: pass

    def _salva_memoria_genetica(self):
        try:
            with open(MEMORIA_FILE, "w") as f: json.dump({"battery_kwh": self.battery_kwh}, f)
        except: pass

    def processa_ciclo_transitorio(self, dati_iot):
        t = datetime.now().strftime('%H:%M:%S')
        
        hp, eff, vero_rpm, vero_psi = self.motore_fisico.elabora_ciclo_fisico(
            dati_iot["fluido"], dati_iot["rpm"], dati_iot["temp_h"], 
            dati_iot["temp_c"], dati_iot["pressione"], dati_iot["is_online"]
        )
        
        surplus_kw = (hp * 0.7457) * (eff / 100.0) * 0.085 
        self.battery_kwh += (surplus_kw / 3600)
        if surplus_kw > 0: self._salva_memoria_genetica() 
        
        if vero_rpm == 0.0:
            msg = f"[{t}] 🔴 [TOTAL S]: Rotore fermo. Motore Stirling disattivato."
            return hp, eff, vero_rpm, vero_psi, msg, (1, 0, 0, 1) 
        elif vero_rpm < 1980.0:
            msg = f"[{t}] ⚡ [REGIME INERZIALE]: Inerzia {vero_rpm} RPM. Freno Meccanico in azione."
            return hp, eff, vero_rpm, vero_psi, msg, (1, 1, 0, 1) 
        else:
            msg = f"[{t}] 🎯 [STABILIZAZIONE]: Allineamento Perfetto ({self.resonance_hz} Hz)."
            return hp, eff, vero_rpm, vero_psi, msg, (0, 1, 0.8, 1) 

class SensoreIoT_Cloud:
    def __init__(self, endpoint_url="http://127.0.0.1:8000"):
        self.endpoint_url = endpoint_url
        self.sensor_data = queue.Queue(maxsize=2)
        self.active = True
        threading.Thread(target=self._hardware_loop, daemon=True).start()

    def _hardware_loop(self):
        idx = 0
        while self.active:
            dati = None
            try:
                risposta = requests.get(self.endpoint_url, timeout=1.5)
                if risposta.status_code == 200:
                    dati_reali = risposta.json()
                    dati = {
                        "fluido": dati_reali.get("fluid", "Hydrogen"),
                        "pressione": float(dati_reali.get("psi", 1000.0)),
                        "rpm": float(dati_reali.get("rpm", 2000.0)),
                        "temp_h": float(dati_reali.get("t_heater", 1450.0)),
                        "temp_c": float(dati_reali.get("t_cooler", 75.0)),
                        "is_online": True
                    }
            except requests.exceptions.RequestException:
                pass
            
            if not dati:
                dati = {
                    "fluido": "Hydrogen [RETE OFFLINE]", 
                    "pressione": 0.0, "rpm": 0.0,
                    "temp_h": 0.0, "temp_c": 0.0,
                    "is_online": False
                }

            if self.sensor_data.full(): self.sensor_data.get_nowait()
            self.sensor_data.put(dati)
            
            idx += 1
            if idx % 15 == 0: scan_anti_sniffing()
            time.sleep(1.0)

    def get_dati(self):
        try: return self.sensor_data.get_nowait()
        except queue.Empty: return None
